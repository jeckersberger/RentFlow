```go
	return events.NewClient(cfg.EventStore.URL, cfg.EventStore.Credentials)

	wire.Build(
		newPgxPool,
		newRedisClient,
		newEventClient,
		initInventoryService,
		newHealthChecker,
		newHTTPServer,
	)
	return nil, nil
```

**cmd/server/main.go:**

```go
package main

import (
	"context"
	"fmt"
	"log"
	"net/http"
	"os"
	"os/signal"
	"syscall"
	"time"

	"rentflow/pkg/common/config"
)

type Server struct {
	httpServer *http.Server
	svc        *application.InventoryService
	health     *health.Checker
}

func main() {
	// 1. Config laden
	cfg, err := config.LoadConfig("config.yaml", "INVENTORY_")
	if err != nil {
		log.Fatalf("Failed to load config: %v", err)
	}

	// 2. Dependency Injection (Wire)
	server, err := initServer(cfg)
	if err != nil {
		log.Fatalf("Failed to initialize server: %v", err)
	}

	// 3. Graceful Shutdown Setup
	sigChan := make(chan os.Signal, 1)
	signal.Notify(sigChan, syscall.SIGINT, syscall.SIGTERM)

	// 4. Start HTTP Server
	go func() {
		log.Printf("Starting inventory-service on %s", cfg.Server.Addr)
		if err := server.httpServer.ListenAndServe(); err != nil && err != http.ErrServerClosed {
			log.Fatalf("HTTP server error: %v", err)
		}
	}()

	// 5. Wait for Shutdown Signal
	<-sigChan
	log.Println("Shutdown signal received, graceful shutdown initiated...")

	// 6. Graceful Shutdown (30s timeout)
	ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
	defer cancel()

	if err := server.httpServer.Shutdown(ctx); err != nil {
		log.Printf("HTTP server shutdown error: %v", err)
	}

	log.Println("Server stopped")
}

func newHTTPServer(handler *adapters.HTTPHandler, health *health.Checker) *http.Server {
	mux := http.NewServeMux()

	// Health Endpoints
	mux.HandleFunc("GET /health", health.HealthHandler)
	mux.HandleFunc("GET /ready", health.ReadyHandler)

	// Business Endpoints
	mux.HandleFunc("POST /equipment", handler.CreateEquipment)
	mux.HandleFunc("GET /equipment/{id}", handler.GetEquipment)
	mux.HandleFunc("PUT /equipment/{id}", handler.UpdateEquipment)
	mux.HandleFunc("DELETE /equipment/{id}", handler.DeleteEquipment)

	return &http.Server{
		Addr:         ":8002",
		Handler:      mux,
		ReadTimeout:  15 * time.Second,
		WriteTimeout: 15 * time.Second,
		IdleTimeout:  60 * time.Second,
	}
}

func newHealthChecker(pgPool *pgxpool.Pool, eventClient *events.Client) *health.Checker {
	return health.NewChecker(
		health.WithPostgresCheck(pgPool),
		health.WithKurrentDBCheck(eventClient),
	)
}
```

### 1.4 Application Service

**internal/application/service.go:**

```go
package application

import (
	"context"
	"fmt"
	"rentflow/internal/domain"
	"rentflow/internal/ports"
)

type InventoryService struct {
	eventStore ports.EventStore
	equipRepo  ports.EquipmentRepository
	cache      ports.CacheLayer
}

func NewInventoryService(
	eventStore ports.EventStore,
	equipRepo ports.EquipmentRepository,
	cache ports.CacheLayer,
) *InventoryService {
	return &InventoryService{
		eventStore: eventStore,
		equipRepo:  equipRepo,
		cache:      cache,
	}
}

// CreateEquipment — Command Handler
func (s *InventoryService) CreateEquipment(
	ctx context.Context,
	cmd ports.CreateEquipmentCommand,
) (*ports.CreateEquipmentResponse, error) {
	// 1. Create Aggregate (Domain Logic)
	equipment, err := domain.NewEquipment(
		cmd.TenantID,
		cmd.Name,
		cmd.Description,
		cmd.Category,
		cmd.Price,
	)
	if err != nil {
		return nil, fmt.Errorf("invalid equipment: %w", err)
	}

	// 2. Publish EquipmentCreated Event to KurrentDB
	event := domain.EquipmentCreated{
		EquipmentID: equipment.ID,
		TenantID:    equipment.TenantID,
		Name:        equipment.Name,
		Category:    equipment.Category,
		Price:       equipment.Price,
		CreatedAt:   time.Now(),
	}

	streamID := fmt.Sprintf("equipment-%s", equipment.ID)
	if err := s.eventStore.AppendEvent(ctx, streamID, event, 0); err != nil {
		return nil, fmt.Errorf("failed to append event: %w", err)
	}

	// 3. Save to Read-Model (PostgreSQL)
	if err := s.equipRepo.Save(ctx, equipment); err != nil {
		return nil, fmt.Errorf("failed to save read-model: %w", err)
	}

	// 4. Invalidate Cache
	s.cache.Delete(ctx, fmt.Sprintf("equipment:%s", equipment.ID))

	return &ports.CreateEquipmentResponse{
		ID: equipment.ID,
		OK: true,
	}, nil
}

// GetEquipment — Query Handler
func (s *InventoryService) GetEquipment(
	ctx context.Context,
	query ports.GetEquipmentQuery,
) (*ports.GetEquipmentResponse, error) {
	// 1. Try Cache
	var cached ports.GetEquipmentResponse
	cacheKey := fmt.Sprintf("equipment:%s", query.EquipmentID)
	if err := s.cache.Get(ctx, cacheKey, &cached); err == nil {
		return &cached, nil
	}

	// 2. Query Read-Model
	equipment, err := s.equipRepo.FindByID(ctx, query.EquipmentID)
	if err != nil {
		return nil, fmt.Errorf("equipment not found: %w", err)
	}

	// 3. Tenant Isolation Check
	if equipment.TenantID != query.TenantID {
		return nil, fmt.Errorf("unauthorized: equipment belongs to different tenant")
	}

	// 4. Build Response
	resp := &ports.GetEquipmentResponse{
		ID:          equipment.ID,
		Name:        equipment.Name,
		Category:    equipment.Category,
		Price:       equipment.Price,
		LastUpdated: equipment.UpdatedAt.Format(time.RFC3339),
	}

	// 5. Cache Response (5 min TTL)
	s.cache.Set(ctx, cacheKey, resp, 5*time.Minute)

	return resp, nil
}
```

```go
package http

import (
	"encoding/json"
	"errors"
	"net/http"
	"rentflow/internal/ports"
	"rentflow/pkg/common/auth"
	"rentflow/pkg/common/errors"
	"rentflow/pkg/common/middleware"
)

type HTTPHandler struct {
	createEquipmentUC ports.CreateEquipmentUseCase
	getEquipmentUC    ports.GetEquipmentUseCase
}

func NewHTTPHandler(
	createEquipmentUC ports.CreateEquipmentUseCase,
	getEquipmentUC ports.GetEquipmentUseCase,
) *HTTPHandler {
	return &HTTPHandler{
		createEquipmentUC: createEquipmentUC,
		getEquipmentUC:    getEquipmentUC,
	}
}

// POST /equipment
func (h *HTTPHandler) CreateEquipment(w http.ResponseWriter, r *http.Request) {
	// 1. Extract Tenant from JWT Claims
	claims, ok := auth.ClaimsFromContext(r.Context())
	if !ok {
		writeError(w, http.StatusUnauthorized, "missing credentials")
		return
	}

	// 2. Parse Request Body
	var req struct {
		Name        string  `json:"name"`
		Description string  `json:"description"`
		Category    string  `json:"category"`
		Price       float64 `json:"price"`
	}
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		writeError(w, http.StatusBadRequest, "invalid request body")
		return
	}

	// 3. Execute Command
	cmd := ports.CreateEquipmentCommand{
		TenantID:    claims.TenantID,
		Name:        req.Name,
		Description: req.Description,
		Category:    req.Category,
		Price:       req.Price,
	}

	resp, err := h.createEquipmentUC.Execute(r.Context(), cmd)
	if err != nil {
		if errors.IsDomainError(err) {
			writeError(w, http.StatusBadRequest, err.Error())
		} else {
			writeError(w, http.StatusInternalServerError, "internal error")
		}
		return
	}

	// 4. Write Response
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(http.StatusCreated)
	json.NewEncoder(w).Encode(resp)
}

// GET /equipment/{id}
func (h *HTTPHandler) GetEquipment(w http.ResponseWriter, r *http.Request) {
	claims, ok := auth.ClaimsFromContext(r.Context())
	if !ok {
		writeError(w, http.StatusUnauthorized, "missing credentials")
		return
	}

	equipmentID := r.PathValue("id")
	query := ports.GetEquipmentQuery{
		EquipmentID: equipmentID,
		TenantID:    claims.TenantID,
	}

	resp, err := h.getEquipmentUC.Execute(r.Context(), query)
	if err != nil {
		writeError(w, http.StatusNotFound, "equipment not found")
		return
	}

	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(resp)
}

func writeError(w http.ResponseWriter, status int, message string) {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(status)
	json.NewEncoder(w).Encode(map[string]interface{}{
		"error": message,
	})
}
```

**internal/adapters/http/middleware.go:**

```go
package http

import (
	"context"
	"net/http"
	"rentflow/pkg/common/auth"
	"rentflow/pkg/common/middleware"
)

func AuthMiddleware(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		token := r.Header.Get("Authorization")
		if token == "" {
			http.Error(w, "missing authorization", http.StatusUnauthorized)
			return
		}

		claims, err := auth.ValidateJWT(token)
		if err != nil {
			http.Error(w, "invalid token", http.StatusUnauthorized)
			return
		}

		ctx := auth.WithClaims(r.Context(), claims)
		next.ServeHTTP(w, r.WithContext(ctx))
	})
}

func TenantMiddleware(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		claims, ok := auth.ClaimsFromContext(r.Context())
		if !ok {
			http.Error(w, "no tenant", http.StatusForbidden)
			return
		}

		ctx := context.WithValue(r.Context(), "tenant_id", claims.TenantID)
		next.ServeHTTP(w, r.WithContext(ctx))
	})
}

func RequestIDMiddleware(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		requestID := r.Header.Get("X-Request-ID")
		if requestID == "" {
			requestID = middleware.GenerateRequestID()
		}

		ctx := context.WithValue(r.Context(), "request_id", requestID)
		w.Header().Set("X-Request-ID", requestID)
		next.ServeHTTP(w, r.WithContext(ctx))
	})
}

func ChainMiddleware(handler http.Handler, mws ...func(http.Handler) http.Handler) http.Handler {
	for i := len(mws) - 1; i >= 0; i-- {
		handler = mws[i](handler)
	}
	return handler
}
```

**pkg/common/health/checker.go:**

```go
package health

import (
	"context"
	"encoding/json"
	"fmt"
	"net/http"
	"sync"
	"time"

	"github.com/jackc/pgx/v5/pgxpool"
	"rentflow/pkg/common/events"
)

type Checker struct {
	pgPool      *pgxpool.Pool
	eventClient *events.Client
	mu          sync.RWMutex
	lastCheck   map[string]checkResult
}

type checkResult struct {
	OK        bool
	Timestamp time.Time
	Error     string
}

type HealthResponse struct {
	Status  string            `json:"status"`
	Checks  map[string]string `json:"checks"`
	Message string            `json:"message,omitempty"`
}

func NewChecker(pgPool *pgxpool.Pool, eventClient *events.Client) *Checker {
	return &Checker{
		pgPool:      pgPool,
		eventClient: eventClient,
		lastCheck:   make(map[string]checkResult),
	}
}

// HealthHandler — Liveness Probe (Kubernetes)
func (c *Checker) HealthHandler(w http.ResponseWriter, r *http.Request) {
	ctx, cancel := context.WithTimeout(r.Context(), 5*time.Second)
	defer cancel()

	resp := HealthResponse{
		Status: "ok",
		Checks: make(map[string]string),
	}

	// Check PostgreSQL
	if err := c.pgPool.Ping(ctx); err != nil {
		resp.Status = "degraded"
		resp.Checks["postgres"] = fmt.Sprintf("error: %v", err)
	} else {
		resp.Checks["postgres"] = "ok"
	}

	// Check KurrentDB
	if err := c.eventClient.Ping(ctx); err != nil {
		resp.Status = "degraded"
		resp.Checks["kurrentdb"] = fmt.Sprintf("error: %v", err)
	} else {
		resp.Checks["kurrentdb"] = "ok"
	}

	statusCode := http.StatusOK
	if resp.Status == "degraded" {
		statusCode = http.StatusServiceUnavailable
	}

	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(statusCode)
	json.NewEncoder(w).Encode(resp)
}

// ReadyHandler — Readiness Probe (Kubernetes)
func (c *Checker) ReadyHandler(w http.ResponseWriter, r *http.Request) {
	ctx, cancel := context.WithTimeout(r.Context(), 10*time.Second)
	defer cancel()

	// Check all dependencies are ready
	pgErr := c.pgPool.Ping(ctx)
	eventErr := c.eventClient.Ping(ctx)

	if pgErr != nil || eventErr != nil {
		w.Header().Set("Content-Type", "application/json")
		w.WriteHeader(http.StatusServiceUnavailable)
		json.NewEncoder(w).Encode(map[string]string{
			"status": "not_ready",
			"reason": "dependencies not available",
		})
		return
	}

	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(http.StatusOK)
	json.NewEncoder(w).Encode(map[string]string{
		"status": "ready",
	})
}
```

**Equipment Aggregate:**

```go
package domain

import "errors"

type Equipment struct {
	ID          string
	TenantID    string
	Name        string
	Description string
	Category    string
	Price       float64
	Available   int      // Verfügbare Menge
	Reserved    int      // Reservierte Menge
	Damaged     int      // Beschädigte Menge
	CreatedAt   time.Time
	UpdatedAt   time.Time
	DeletedAt   *time.Time
	Version     int64 // Optimistic Locking
}

// Invarianten:
// 1. Available + Reserved + Damaged >= 0
// 2. Price >= 0
// 3. Name nicht leer
// 4. TenantID nicht leer

func (e *Equipment) Validate() error {
	if e.Name == "" {
		return errors.New("equipment name cannot be empty")
	}
	if e.TenantID == "" {
		return errors.New("tenant_id is required")
	}
	if e.Price < 0 {
		return errors.New("price cannot be negative")
	}
	if (e.Available + e.Reserved + e.Damaged) < 0 {
		return errors.New("invalid stock counts")
	}
	return nil
}

func (e *Equipment) Reserve(quantity int) error {
	if quantity > e.Available {
		return errors.New("insufficient stock")
	}
	e.Available -= quantity
	e.Reserved += quantity
	return nil
}

func (e *Equipment) Release(quantity int) error {
	if quantity > e.Reserved {
		return errors.New("cannot release more than reserved")
	}
	e.Reserved -= quantity
	e.Available += quantity
	return nil
}

func (e *Equipment) MarkDamaged(quantity int) error {
	if quantity > e.Available {
		return errors.New("insufficient stock to mark as damaged")
	}
	e.Available -= quantity
	e.Damaged += quantity
	return nil
}
```

```go
package events

import (
	"encoding/json"
	"fmt"
)

type VersionedEvent struct {
	Type    string          `json:"type"`
	Version int             `json:"version"`
	Data    json.RawMessage `json:"data"`
	Meta    EventMetadata   `json:"meta"`
}

type EventMetadata struct {
	EventID       string `json:"event_id"`
	CorrelationID string `json:"correlation_id"`
	Timestamp     int64  `json:"timestamp"`
	TenantID      string `json:"tenant_id"`
}

// Upfrongrade alte Events auf aktuelle Version
func (ve *VersionedEvent) Upgrade() error {
	switch ve.Type {
	case "EquipmentCreated":
		return ve.upgradeEquipmentCreated()
	case "InvoiceCreated":
		return ve.upgradeInvoiceCreated()
	default:
		return nil // Unbekannte Events werden ignoriert
	}
}

func (ve *VersionedEvent) upgradeEquipmentCreated() error {
	if ve.Version == 2 {
		return nil // Aktuelle Version
	}

	if ve.Version == 1 {
		// Parse v1
		var v1 struct {
			EquipmentID string  `json:"equipment_id"`
			TenantID    string  `json:"tenant_id"`
			Name        string  `json:"name"`
			Category    string  `json:"category"`
			Price       float64 `json:"price"`
		}

		if err := json.Unmarshal(ve.Data, &v1); err != nil {
			return fmt.Errorf("failed to parse v1: %w", err)
		}

		// Upgrade zu v2
		v2 := struct {
			EquipmentID string  `json:"equipment_id"`
			TenantID    string  `json:"tenant_id"`
			Name        string  `json:"name"`
			Description string  `json:"description"`
			Category    string  `json:"category"`
			Price       float64 `json:"price"`
		}{
			EquipmentID: v1.EquipmentID,
			TenantID:    v1.TenantID,
			Name:        v1.Name,
			Description: "", // Default
			Category:    v1.Category,
			Price:       v1.Price,
		}

		data, err := json.Marshal(v2)
		if err != nil {
			return fmt.Errorf("failed to marshal v2: %w", err)
		}

		ve.Data = data
		ve.Version = 2
		return nil
	}

	return fmt.Errorf("unknown version: %d", ve.Version)
}
```

### 2.7 Idempotenz & at-least-once Delivery

```go
// Jedes Event hat eindeutige event_id
// Jeder Service speichert "processed_events" Tabelle
type ProcessedEvent struct {
	TenantID      string
	EventID       string // PK für Idempotenz
	EventType     string
	ProcessedAt   time.Time
	CorrelationID string
}

// Handler checkt zuerst ob Event bereits verarbeitet wurde
func (ps *ProjectionService) handleEquipmentCreated(
	ctx context.Context,
	event *domain.EquipmentCreated,
) error {
	// 1. Check if already processed (idempotent)
	query := `SELECT event_id FROM inventory_schema.processed_events WHERE event_id = $1`
	var existingID string
	err := ps.pgPool.QueryRow(ctx, query, event.EventID).Scan(&existingID)
	if err == nil {
		// Already processed, skip
		return nil
	}
	if err != pgx.ErrNoRows {
		return fmt.Errorf("check failed: %w", err)
	}

	// 2. Process event
	tx, err := ps.pgPool.Begin(ctx)
	if err != nil {
		return err
	}
	defer tx.Rollback(ctx)

	// 2a. Insert into read-model
	insertQuery := `
	INSERT INTO inventory_schema.equipment
	(id, tenant_id, name, description, category, price, created_at)
	VALUES ($1, $2, $3, $4, $5, $6, $7)
	`
	_, err = tx.Exec(ctx, insertQuery,
		event.EquipmentID,
		event.TenantID,
		event.Name,
		event.Description,
		event.Category,
		event.Price,
		event.CreatedAt,
	)
	if err != nil {
		return fmt.Errorf("insert failed: %w", err)
	}

	// 2b. Mark as processed
	processedQuery := `
	INSERT INTO inventory_schema.processed_events
	(tenant_id, event_id, event_type, processed_at)
	VALUES ($1, $2, $3, $4)
	`
	_, err = tx.Exec(ctx, processedQuery,
		event.TenantID,
		event.EventID,
		event.EventType(),
		time.Now(),
	)
	if err != nil {
		return fmt.Errorf("mark processed failed: %w", err)
	}

	// 3. Commit transaction
	return tx.Commit(ctx)
}
```

### 2.9 Fehlerbehandlung & Retry-Strategie

```go
package infrastructure

import (
	"context"
	"fmt"
	"time"
)

type RetryConfig struct {
	MaxAttempts      int           // 3 Versuche
	InitialBackoff   time.Duration // 100ms
	MaxBackoff       time.Duration // 30s
	BackoffMultiplier float64      // 2x exponentiell
}

type SubscriptionHandler struct {
	retryConfig RetryConfig
	deadLetterQ chan DeadLetterMessage
}

type DeadLetterMessage struct {
	EventID       string
	EventType     string
	TenantID      string
	Error         string
	LastAttempt   time.Time
	Attempts      int
	CorrelationID string
}

// HandleEvent mit Retry
func (sh *SubscriptionHandler) HandleEvent(
	ctx context.Context,
	event *events.Event,
) error {
	var lastErr error
	backoff := sh.retryConfig.InitialBackoff

	for attempt := 1; attempt <= sh.retryConfig.MaxAttempts; attempt++ {
		err := sh.processEvent(ctx, event)
		if err == nil {
			return nil
		}

		lastErr = err
		if attempt < sh.retryConfig.MaxAttempts {
			select {
			case <-ctx.Done():
				return fmt.Errorf("context cancelled: %w", lastErr)
			case <-time.After(backoff):
				backoff = time.Duration(float64(backoff) * sh.retryConfig.BackoffMultiplier)
				if backoff > sh.retryConfig.MaxBackoff {
					backoff = sh.retryConfig.MaxBackoff
				}
			}
		}
	}

	// Alle Versuche fehlgeschlagen → Dead Letter Queue
	sh.sendToDeadLetterQueue(DeadLetterMessage{
		EventID:       event.ID,
		EventType:     event.Type,
		TenantID:      event.TenantID,
		Error:         lastErr.Error(),
		LastAttempt:   time.Now(),
		Attempts:      sh.retryConfig.MaxAttempts,
		CorrelationID: event.CorrelationID,
	})

	return fmt.Errorf("all retry attempts failed: %w", lastErr)
}

func (sh *SubscriptionHandler) sendToDeadLetterQueue(msg DeadLetterMessage) {
	select {
	case sh.deadLetterQ <- msg:
	default:
		// Dead Letter Queue voll, loggen
		log.Printf("Dead letter queue full: %+v", msg)
	}
}

// KurrentDB Down Handling
func (sh *SubscriptionHandler) StartSubscriptionWithRestart(
	ctx context.Context,
	eventClient *events.Client,
) {
	for {
		err := eventClient.Subscribe(ctx, "$ce-equipment", sh.HandleEvent)
		if err != nil {
			log.Printf("Subscription lost: %v, reconnecting in 10s...", err)
			time.Sleep(10 * time.Second)
			continue
		}
		break
	}
}
```

```sql
CREATE MATERIALIZED VIEW project_schema.active_projects_summary AS
SELECT
	p.id,
	p.tenant_id,
	p.name,
	p.start_date,
	p.end_date,
	(SELECT COUNT(*) FROM project_schema.packlists pl WHERE pl.project_id = p.id) AS items_count,
	(SELECT COUNT(DISTINCT rs.equipment_id)
	 FROM project_schema.packlists pl
	 JOIN inventory_schema.reservations rs ON rs.project_id = pl.project_id)
	AS equipment_count,
	p.updated_at
FROM project_schema.projects p
WHERE p.status IN ('active', 'planning')
	AND p.tenant_id = current_setting('app.current_tenant_id')::uuid;
```
